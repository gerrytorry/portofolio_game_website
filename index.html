Item Recall System (Lua)

-- Roblox Studio: Recall System Script by Lhick (@babylootion)
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local RecallEvent = ReplicatedStorage:WaitForChild("RecallEvent")

local function OnRecallTriggered(player: Player, checkpointCFrame: CFrame)
    local character = player.Character
    if character and character:FindFirstChild("HumanoidRootPart") then
        character.HumanoidRootPart.CFrame = checkpointCFrame
        print(`Player {player.Name} recalled to checkpoint successfully!`)
    end
end

RecallEvent.OnServerEvent:Connect(OnRecallTriggered)
